cmake_minimum_required(VERSION 3.17)
project(acorn_permission_adapter LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenMP REQUIRED)
if(NOT ACORN_SOURCE_DIR OR NOT NLOHMANN_SOURCE_DIR)
  message(FATAL_ERROR "Fixed author ACORN and JSON source directories are required")
endif()
set(FAISS_ENABLE_GPU OFF CACHE BOOL "" FORCE)
set(FAISS_ENABLE_PYTHON OFF CACHE BOOL "" FORCE)
set(FAISS_ENABLE_C_API OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(FAISS_OPT_LEVEL generic CACHE STRING "" FORCE)
set(FETCHCONTENT_SOURCE_DIR_NLOHMANN_JSON "${NLOHMANN_SOURCE_DIR}" CACHE PATH "" FORCE)
set(JSON_BuildTests OFF CACHE BOOL "" FORCE)
set(JSON_Install OFF CACHE BOOL "" FORCE)
add_subdirectory("${ACORN_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/author" EXCLUDE_FROM_ALL)
add_executable(acorn_port acorn_port.cpp)
target_include_directories(acorn_port PRIVATE "${ACORN_SOURCE_DIR}" "${NLOHMANN_SOURCE_DIR}/include" "${NLOHMANN_SOURCE_DIR}/single_include")
target_link_libraries(acorn_port PRIVATE faiss OpenMP::OpenMP_CXX ${BLAS_LIBRARIES} ${LAPACK_LIBRARIES} ${ZLIB_LIBRARY})
